users {
	id integer pk increments unique
	user_name varchar(255) unique
	pass varchar(255)
	police_id varchar(255) unique
	full_name varchar(255)
	email varchar(255) unique
	location varchar(255)
	role integer >* roles.id
}

tasks {
	id integer pk increments unique
	title varchar(255)
	user_id integer >* users.id
	created_at timestamp
	due_at date null
	is_done tinyint def(0)
	notes text null
}

tasks_item {
	id integer pk increments unique
	task_id integer >* tasks.id
	description varchar(255) null
	is_done tinyint(10) def(0)
}

suspects {
	id integer pk increments unique
	name varchar(255) null
	sus_id varchar(255) null unique
	last_seen text
	last_seen_date date
	pela_id varchar(255)
	user_id integer >* users.id
	created_at timestamp
	location varchar(255)
}

roles {
	id integer pk increments unique
	name varchar(255) unique
}

tasks_attachments {
	id integer pk increments unique
	task_id integer >* tasks.id
	file_path varchar(255)
	file_name varchar(255)
	uploaded_at timestamp
}

sus_attachments {
	id integer pk increments unique
	sus_id integer >* suspects.id
	file_path varchar(255)
	file_name varchar(255)
	uploaded_at timestamp
}
